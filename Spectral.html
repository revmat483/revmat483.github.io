// 0.03 Spectral — отправка уведомлений ТОЛЬКО ДЛЯ ТВОЕГО ID
// Perfect Click + Firebase Realtime Database
// ============================================================
// 1. Включи "Расширенные возможности Java" в Perfect Click
// 2. Скопируй свой USER_ID с сайта (кнопка "Копировать ID")
// 3. Вставь ID в переменную ниже (вместо "СКОПИРУЙ_СВОЙ_ID")
// ============================================================

import pfc;

// ⚡ ВАЖНО: Вставь сюда свой персональный ID (скопированный с сайта)
String USER_ID = "СКОПИРУЙ_СВОЙ_ID";   // 🔁 ЗАМЕНИТЕ НА ВАШ ID

void sendPersonalNotification(String text) {
    // База данных защищена, endpoint известен только скрипту
    String baseUrl = "https://gamep-79cef-default-rtdb.firebaseio.com/users/" + USER_ID + "/notifications.json";
    
    String currentTime = Time.getTime();
    long uniqueId = System.currentTimeMillis();
    
    String safeText = text.replace("\"", "\\\"").replace("\n", " ");
    
    String json = "{"
        + "\"text\":\"" + safeText + "\","
        + "\"time\":\"" + currentTime + "\","
        + "\"timestamp\":" + uniqueId
        + "}";
    
    pfc.log("[0.03 Spectral] Отправка для ID: " + USER_ID);
    pfc.log("Текст: " + text);
    
    try {
        String response = pfc.execJsonPOST(baseUrl, json);
        pfc.log("✅ Уведомление отправлено в персональный канал!");
    } catch (Exception e) {
        pfc.log("❌ Ошибка: " + e.toString());
        pfc.log("Проверь интернет и правильность USER_ID");
    }
}

// ============= ПРИМЕРЫ ВЫЗОВА =============
